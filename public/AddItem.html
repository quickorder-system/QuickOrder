function previewImage(event) { const preview = document.getElementById('preview'); const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(ev) { preview.src = ev.target.result; preview.style.display = 'block'; document.getElementById('itemImageData').value = ev.target.result; }; reader.readAsDataURL(file); } document.getElementById('itemPhoto')?.addEventListener('change', previewImage); // Redirect to Admin.html or Owner.html after form submission document.querySelector('.form-container form').addEventListener('submit', function(e) { e.preventDefault(); // Save item to localStorage const item = { itemName: document.getElementById('itemName').value, category: document.getElementById('category').value, unit: document.getElementById('unit').value, quantity: parseFloat(document.getElementById('quantity').value), alertLevel: parseFloat(document.getElementById('alertLevel').value), description: document.getElementById('description').value, image: document.getElementById('itemImageData')?.value || '' }; let items = JSON.parse(localStorage.getItem('inventoryItems') || '[]'); items.push(item); localStorage.setItem('inventoryItems', JSON.stringify(items)); // Check last user type const lastUser = localStorage.getItem('lastUserType') || 'admin'; if (lastUser === 'owner') { window.location.href = 'Owner.html'; } else { window.location.href = 'Admin.html'; } }); // Set lastUserType when coming from Admin or Owner if (document.referrer.includes('Owner.html')) { localStorage.setItem('lastUserType', 'owner'); } else if (document.referrer.includes('Admin.html')) { localStorage.setItem('lastUserType', 'admin'); }
